from flask import Flask, jsonify, requestfrom flask_cors import CORSimport mysql.connectorimport pandas as pdapp = Flask(__name__)CORS(app)def get_db_connection():    return mysql.connector.connect(        host="localhost",        user="root",        password="",        database="dumbo_control_financiero"    )@app.route('/api/cuentas-por-cobrar', methods=['GET'])def get_cuentas():    conn = get_db_connection()    cursor = conn.cursor(dictionary=True)    cursor.execute("SELECT * FROM cuentas_por_cobrar")    datos = cursor.fetchall()    cursor.close()    conn.close()    return jsonify(datos)@app.route('/api/exportar-excel', methods=['GET'])def exportar_excel():    conn = get_db_connection()    query = "SELECT * FROM cuentas_por_cobrar"    df = pd.read_sql(query, conn)    conn.close()        df.to_excel("reporte_dumbo_financiero.xlsx", index=False)    return jsonify({"mensaje": "Archivo reporte_dumbo_financiero.xlsx generado exitosamente"})if __name__ == '__main__':    app.run(port=5000, debug=True)
