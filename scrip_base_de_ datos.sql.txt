CREATE TABLE cuentas_por_cobrar (id INT AUTO_INCREMENT PRIMARY KEY,cliente VARCHAR(150) NOT NULL,monto DECIMAL(10,2) NOT NULL,fecha_emision DATE NOT NULL,fecha_vencimiento DATE NOT NULL,estado VARCHAR(30) NOT NULL,pagos_realizados DECIMAL(10,2) NOT NULL DEFAULT 0.00,concepto VARCHAR(255) NULL,saldo_pendiente DECIMAL(10,2) NOT NULL,observaciones TEXT NULL);CREATE TABLE cuentas_por_pagar (id INT AUTO_INCREMENT PRIMARY KEY,proveedor VARCHAR(150) NOT NULL,monto DECIMAL(10,2) NOT NULL,fecha DATE NOT NULL,vencimiento DATE NOT NULL,estado VARCHAR(30) NOT NULL,pagos DECIMAL(10,2) NOT NULL DEFAULT 0.00,concepto VARCHAR(255) NULL,saldo_pendiente DECIMAL(10,2) NOT NULL,observaciones TEXT NULL);CREATE TABLE control_tributario (id INT AUTO_INCREMENT PRIMARY KEY,obligacion_tributaria VARCHAR(150) NOT NULL,tipo_tributo VARCHAR(100) NOT NULL,periodo VARCHAR(20) NOT NULL,fecha_limite DATE NOT NULL,estado VARCHAR(30) NOT NULL,monto_estimado DECIMAL(10,2) NULL,fecha_cumplimiento DATE NULL,responsable VARCHAR(150) NULL,observaciones TEXT NULL);
